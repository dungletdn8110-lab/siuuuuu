import React, { useState, useEffect } from 'react';
import { socket } from './socket';
import { RoomState, Role } from '../shared/types';
import ModeratorView from './components/ModeratorView';
import TeamView from './components/TeamView';
import ProjectorView from './components/ProjectorView';
import RuleModal from './components/RuleModal';
import { Shield, Crown, Anchor, Tv, BookOpen } from 'lucide-react';

export default function App() {
  const [joined, setJoined] = useState(false);
  const [roomCode, setRoomCode] = useState('SIAM1893');
  const [role, setRole] = useState<Role>('SPECTATOR');
  const [roomState, setRoomState] = useState<RoomState | null>(null);
  const [showRules, setShowRules] = useState(false);

  useEffect(() => {
    // Khôi phục phòng và role từ LocalStorage nếu vô tình F5
    const savedRoom = localStorage.getItem('siam_room');
    const savedRole = localStorage.getItem('siam_role') as Role;
    if (savedRoom && savedRole) {
      setRoomCode(savedRoom);
      setRole(savedRole);
      socket.emit('JOIN_ROOM', { roomCode: savedRoom, role: savedRole });
      setJoined(true);
    }

    socket.on('ROOM_STATE_UPDATE', (state: RoomState) => {
      setRoomState(state);
    });

    return () => {
      socket.off('ROOM_STATE_UPDATE');
    };
  }, []);

  const handleJoin = (selectedRole: Role) => {
    const code = roomCode.toUpperCase().trim();
    if (!code) return;
    setRole(selectedRole);
    localStorage.setItem('siam_room', code);
    localStorage.setItem('siam_role', selectedRole);
    socket.emit('JOIN_ROOM', { roomCode: code, role: selectedRole });
    setJoined(true);
  };

  if (!joined || !roomState) {
    return (
      <div className="min-h-screen bg-[#F5F7FA] text-[#172033] flex flex-col justify-center items-center p-4">
        <div className="max-w-xl w-full bg-white rounded-2xl shadow-xl p-8 border border-slate-200">
          <div className="text-center mb-8">
            <h1 className="text-3xl font-extrabold tracking-tight text-slate-900 mb-2">
              BÀN CỜ ĐÔNG NAM Á
            </h1>
            <h2 className="text-xl font-semibold text-[#27856A] mb-3">VẬN MỆNH XIÊM</h2>
            <p className="text-sm text-slate-600 leading-relaxed">
              Trò chơi chiến thuật lịch sử mô phỏng cuộc cạnh tranh ảnh hưởng của Anh & Pháp cuối thế kỉ XIX và chính sách ngoại giao bảo vệ độc lập của Xiêm.
            </p>
          </div>

          <div className="space-y-4 mb-6">
            <div>
              <label className="block text-xs font-bold text-slate-700 uppercase tracking-wider mb-1">Mã Phòng Chơi</label>
              <input
                type="text"
                value={roomCode}
                onChange={e => setRoomCode(e.target.value.toUpperCase())}
                className="w-full px-4 py-3 text-lg font-bold tracking-widest text-center border-2 border-slate-300 rounded-xl focus:border-[#27856A] outline-none"
                placeholder="VD: SIAM1893"
              />
            </div>

            <div className="grid grid-cols-1 md:grid-cols-2 gap-3 pt-2">
              <button
                onClick={() => handleJoin('BRITAIN')}
                className="flex items-center justify-center gap-2 p-3.5 rounded-xl border-2 border-[#315C9B] bg-blue-50 text-[#315C9B] font-bold hover:bg-blue-100 transition"
              >
                <Crown size={20} /> Đội Anh (Britain)
              </button>

              <button
                onClick={() => handleJoin('FRANCE')}
                className="flex items-center justify-center gap-2 p-3.5 rounded-xl border-2 border-[#9B3D4D] bg-red-50 text-[#9B3D4D] font-bold hover:bg-red-100 transition"
              >
                <Anchor size={20} /> Đội Pháp (France)
              </button>

              <button
                onClick={() => handleJoin('SIAM')}
                className="flex items-center justify-center gap-2 p-3.5 rounded-xl border-2 border-[#27856A] bg-emerald-50 text-[#27856A] font-bold hover:bg-emerald-100 transition md:col-span-2"
              >
                <Shield size={20} /> Vương Quốc Xiêm (Siam)
              </button>

              <button
                onClick={() => handleJoin('MODERATOR')}
                className="flex items-center justify-center gap-2 p-3 rounded-xl border border-amber-500 bg-amber-50 text-amber-800 font-medium hover:bg-amber-100 transition"
              >
                👑 Quản trò (Host)
              </button>

              <button
                onClick={() => handleJoin('SPECTATOR')}
                className="flex items-center justify-center gap-2 p-3 rounded-xl border border-slate-400 bg-slate-50 text-slate-700 font-medium hover:bg-slate-100 transition"
              >
                <Tv size={18} /> Máy Chiếu Lớp Học
              </button>
            </div>
          </div>

          <div className="text-center">
            <button
              onClick={() => setShowRules(true)}
              className="text-xs font-semibold text-slate-500 hover:text-slate-800 inline-flex items-center gap-1"
            >
              <BookOpen size={14} /> Xem tóm tắt luật chơi & tính điểm
            </button>
          </div>
        </div>

        {showRules && <RuleModal onClose={() => setShowRules(false)} />}
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-[#F5F7FA] text-[#172033] flex flex-col">
      {/* Top Navbar */}
      <header className="bg-white border-b border-slate-200 px-6 py-3 flex items-center justify-between shadow-sm">
        <div className="flex items-center gap-4">
          <span className="font-extrabold text-lg text-[#27856A]">VẬN MỆNH XIÊM</span>
          <span className="bg-slate-100 text-slate-700 text-xs px-2.5 py-1 rounded-md font-mono font-bold">
            PHÒNG: {roomState.roomCode}
          </span>
          <span className="text-xs font-semibold px-2.5 py-1 rounded-md bg-emerald-100 text-emerald-800">
            Lượt {roomState.round}/3
          </span>
        </div>

        <div className="flex items-center gap-3">
          <button
            onClick={() => setShowRules(true)}
            className="text-xs font-medium bg-slate-100 hover:bg-slate-200 px-3 py-1.5 rounded-lg flex items-center gap-1 text-slate-700"
          >
            <BookOpen size={14} /> Luật Chơi
          </button>
          <button
            onClick={() => {
              localStorage.clear();
              setJoined(false);
            }}
            className="text-xs text-red-600 hover:text-red-800 font-medium"
          >
            Đổi vai trò
          </button>
        </div>
      </header>

      {/* Main View theo Role */}
      <main className="flex-1 p-4 md:p-6 max-w-7xl mx-auto w-full">
        {role === 'MODERATOR' && <ModeratorView state={roomState} />}
        {role === 'SPECTATOR' && <ProjectorView state={roomState} />}
        {(role === 'BRITAIN' || role === 'FRANCE' || role === 'SIAM') && (
          <TeamView team={role} state={roomState} />
        )}
      </main>

      {showRules && <RuleModal onClose={() => setShowRules(false)} />}
    </div>
  );
}
